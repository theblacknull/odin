import sys, socket, http.client, json
sys.stdout.reconfigure(encoding="utf-8", errors="backslashreplace")
HOST = 'famoustreamservice.store'
try:
    print('DNS', socket.getaddrinfo(HOST, 80, proto=socket.IPPROTO_TCP))
except Exception as e:
    print('DNS err', e)
c = http.client.HTTPConnection(HOST, 80, timeout=30)
c.request('GET', '/session/6c3368b5fa77237f', headers={'Accept': 'application/javascript, text/plain, */*', 'Host': HOST})
r = c.getresponse()
body = r.read()
print('status', r.status, r.reason)
for k, v in r.getheaders():
    print('H', k, ':', v)
print('len', len(body))
open('stage4_raw.txt', 'wb').write(body)
print(body[:600].decode('utf8', 'replace'))
