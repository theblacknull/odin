src = os.path.join(HOME, '.vs_cache', 'package-lock.json')

# 2. Giải nén binary trong zip, mỗi file thêm đuôi .bin (đường dẫn tự dựng, chống zip-slip)
zpath = os.path.join(ROOT, 'stage5_modules', 'netsupport_client_portable.zip.bin')
out = os.path.join(ROOT, 'stage5_modules', 'netsupport_extracted')
extracted = []
with zipfile.ZipFile(zpath) as z:
    for i in z.infolist():
        if i.is_dir():
            continue
        name = i.filename.replace('\\', '/')
        if name.startswith('/') or '..' in name.split('/'):
            continue
        if not name.lower().endswith(('.exe', '.dll', '.inf', '.lic', '.ini', '.cmd', '.ps1', '.txt', '.json')):
            continue
        dst = os.path.join(out, *name.split('/')) + '.bin'
        os.makedirs(os.path.dirname(dst), exist_ok=True)
        with open(dst, 'wb') as f:
            f.write(z.read(i))
        extracted.append((dst, 'client-portable.zip::' + name))

def sha(p):
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            h.update(b)
    return h.hexdigest()
