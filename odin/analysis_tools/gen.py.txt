"""Generic static deobfuscator for javascript-obfuscator 'string array + rotation + base64' output.
Never executes JS. usage: gen.py in out"""
import sys, re, json, base64, math, urllib.parse
sys.stdout.reconfigure(encoding="utf-8", errors="backslashreplace")
src = open(sys.argv[1], encoding='utf-8', errors='replace').read()

# 1. locate string-array function:  function NAME(){const X=[ ... ];NAME=function(){return X;};return NAME();}
m = re.search(r'function (a0e\w*)\(\)\{const (\w+)=\[', src)
arr_fn = m.group(1)
start = m.end()
end = src.find('];', start)
table = re.findall(r"'((?:[^'\\]|\\.)*)'", src[start:end])
print('array fn', arr_fn, 'size', len(table))

# 2. decoder function: function DEC(e,O){e=e-0xOFF;
md = re.search(r'function (a0e\w*)\((\w+),(\w+)\)\{\2=\2-(0x[0-9a-f]+)', src)
dec_fn, OFF = md.group(1), int(md.group(4), 16)
print('decoder', dec_fn, hex(OFF))
dpart = src[md.start():md.start() + 1500]
alpha = re.search(r"'([A-Za-z0-9+/=]{64,65})'", dpart)
ALPHA = alpha.group(1)
print('alphabet', ALPHA[:12], '...')
rc4 = 'rc4' in dpart.lower() or re.search(r'\[\w+\]\^\w+\[', dpart[:1500]) is not None
print('rc4-like?', bool(re.search(r'%256|\^', dpart)))
STD = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/='
TR = str.maketrans(ALPHA, STD)


def b64(x):
    x = x.translate(TR)
    x += '=' * (-len(x) % 4)
    return base64.b64decode(x)


def dec(x):
    try:
        return urllib.parse.unquote(''.join('%%%02x' % c for c in b64(x)))
    except Exception:
        return None


def pint(v):
    mm = re.match(r'\s*[-+]?\d+', v or '')
    return int(mm.group()) if mm else math.nan


# 3. rotation IIFE: }(ARR,0xTARGET)
mt = re.search(r'\}\(' + re.escape(arr_fn) + r',(0x[0-9a-f]+)\)', src)
target = int(mt.group(1), 16)
k = src.find('const ', src.find('try{', mt.start() - 1800))
mm = re.search(r'try\{const (\w+)=(.*?);if\(\1===\w+\)break', src[max(0, mt.start() - 2000):mt.start()], re.S)
expr = mm.group(2)
expr = re.sub(r'parseInt\(\w+\((0x[0-9a-f]+)\)\)', r'P(\1)', expr)
T = table[:]
for n in range(len(T) + 1):
    P = lambda x: pint(dec(T[x - OFF]))
    try:
        v = eval(expr)
    except Exception:
        v = math.nan
    if v == target:
        print('rotation', n)
        break
    T.append(T.pop(0))
else:
    print('NO ROTATION MATCH')
strs = {hex(OFF + i): dec(v) for i, v in enumerate(T)}
json.dump(strs, open(sys.argv[2] + '.strings.json', 'w'))

# 4. aliases of decoder
al = {dec_fn}
for _ in range(4):
    al |= set(re.findall(r'(?:const|let|var|,)\s*(\w+)\s*=\s*(?:%s)\b(?!\()' % '|'.join(map(re.escape, al)), src))
    al |= set(re.findall(r'\b(\w+)=(?:%s)[,;]' % '|'.join(map(re.escape, al)), src))
print('aliases', len(al))
pat = re.compile(r'\b(%s)\((0x[0-9a-f]+)\)' % '|'.join(map(re.escape, al)))


def sub(mo):
    v = strs.get(mo.group(2))
    return repr(v) if v is not None else mo.group(0)


out = pat.sub(sub, src)
for _ in range(8):
    out = re.sub(r"'((?:[^'\\]|\\.)*)'\s*\+\s*'((?:[^'\\]|\\.)*)'", lambda mo: "'" + mo.group(1) + mo.group(2) + "'", out)
# bracket-notation to dot where simple
out = re.sub(r"\['([A-Za-z_$][\w$]*)'\]", r'.\1', out)
# drop array/decoder definitions
i = out.find('function ' + arr_fn)
open(sys.argv[2], 'w', encoding='utf-8').write(re.sub(r'(;|\{|\})', r'\1\n', out))
print('written', sys.argv[2], len(out))
